# Gesture Playground

Interactive demo of tap, pan, and long-press gestures with composition.

```typescript
import { Component, signal } from '@angular/core';
import {
  Gesture,
  LongPressGesture,
  LynxGestureDetector,
  PanGesture,
  TapGesture,
  LYNX_ELEMENTS,
} from '@blotch/angular-lynx';

@Component({
  selector: 'app-gesture-playground',
  standalone: true,
  imports: [LYNX_ELEMENTS, LynxGestureDetector],
  template: `
    <scroll-view class="container" scroll-orientation="vertical">
      <text class="title">Gesture Playground</text>

      <!-- Tap counter -->
      <text class="section">Tap</text>
      <view class="card" [lynxGestureDetector]="tapGesture">
        <text class="card-text">Tap count: {{ tapCount() }}</text>
        <text class="hint">Tap to increment, double-tap to reset</text>
      </view>

      <!-- Pan tracker -->
      <text class="section">Pan</text>
      <view class="card" [lynxGestureDetector]="panGesture">
        <text class="card-text"> x: {{ panX() }}, y: {{ panY() }} </text>
        <text class="hint">Drag to track position</text>
      </view>

      <!-- Long press -->
      <text class="section">Long Press</text>
      <view
        class="card"
        [class.card-active]="pressing()"
        [lynxGestureDetector]="longPressGesture"
      >
        <text class="card-text">
          {{ pressing() ? 'Pressing...' : 'Hold me' }}
        </text>
        <text class="hint">Hold for 500ms</text>
      </view>

      <!-- Exclusive composition -->
      <text class="section">Exclusive (Pan vs Tap)</text>
      <view class="card" [lynxGestureDetector]="exclusive">
        <text class="card-text">{{ action() }}</text>
        <text class="hint">Pan wins over tap — only one fires</text>
      </view>
    </scroll-view>
  `,
  styles: `
    .container {
      height: 100vh;
      padding: 16px;
      background-color: #f7fafc;
    }
    .title {
      font-size: 28px;
      font-weight: bold;
      color: #1a202c;
      margin-bottom: 16px;
    }
    .section {
      font-size: 18px;
      font-weight: bold;
      color: #2d3748;
      margin-top: 16px;
      margin-bottom: 8px;
    }
    .card {
      background-color: white;
      border-radius: 12px;
      padding: 24px;
      align-items: center;
    }
    .card-active {
      background-color: #ebf8ff;
    }
    .card-text {
      font-size: 20px;
      font-weight: bold;
      color: #2b6cb0;
    }
    .hint {
      font-size: 13px;
      color: #a0aec0;
      margin-top: 8px;
    }
  `,
})
export class GesturePlayground {
  // ── Tap ──
  readonly tapCount = signal(0);

  readonly tapGesture = Gesture.Exclusive(
    new TapGesture().numberOfTaps(2).onEnd(() => this.tapCount.set(0)),
    new TapGesture().onEnd(() => this.tapCount.update((n) => n + 1)),
  );

  // ── Pan ──
  readonly panX = signal(0);
  readonly panY = signal(0);

  readonly panGesture = new PanGesture()
    .minDistance(5)
    .onUpdate((e) => {
      this.panX.set(Math.round(e.translationX));
      this.panY.set(Math.round(e.translationY));
    })
    .onEnd(() => {
      this.panX.set(0);
      this.panY.set(0);
    });

  // ── Long press ──
  readonly pressing = signal(false);

  readonly longPressGesture = new LongPressGesture()
    .minDuration(500)
    .onStart(() => this.pressing.set(true))
    .onEnd(() => this.pressing.set(false));

  // ── Exclusive composition ──
  readonly action = signal('Tap or drag');

  readonly exclusive = Gesture.Exclusive(
    new PanGesture()
      .minDistance(10)
      .onUpdate((e) =>
        this.action.set(
          `Pan x:${Math.round(e.translationX)} y:${Math.round(e.translationY)}`,
        ),
      )
      .onEnd(() => this.action.set('Pan ended')),
    new TapGesture().onEnd(() => this.action.set('Tapped!')),
  );
}
```

## Key concepts

- **`[lynxGestureDetector]`** — Directive that attaches a gesture recognizer to an element
- **`Gesture.Exclusive()`** — Only the first gesture to activate wins; others are canceled
- **`.onUpdate()` / `.onEnd()`** — Callbacks for gesture state changes; update signals to re-render
- **`.minDistance()` / `.minDuration()`** — Configure gesture activation thresholds
